quick_sort.js
const quickSort = (arr) => {
  if (arr.length <= 1) return arr;
  
  let pivot = arr[0];
  let left = arr.slice(1).filter(x => x < pivot);
  let right = arr.slice(1).filter(x => x >= pivot);
  
  return [...quickSort(left), pivot, ...quickSort(right)];
};
